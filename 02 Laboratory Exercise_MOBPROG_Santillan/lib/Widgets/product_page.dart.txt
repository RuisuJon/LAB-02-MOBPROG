class ProductTile extends StatelessWidget {
  const ProductTile({super.key, required this.product});
  final Product product; //Calling the product which the user had chosen

  @override
  Widget build(BuildContext context) {
    //focusing on using the own feature of the dart pad. Also, Build format of the product inputs for the application page
    return Container(
      margin: const EdgeInsets.symmetric(horizontal: 15, vertical: 5),
      decoration: BoxDecoration(
        color: Theme.of(context).cardColor,
        borderRadius: BorderRadius.circular(15),
        border: Border.all(color: Colors.grey.withOpacity(0.5), width: 1),
        boxShadow: [
          BoxShadow(
            color: Colors.black.withOpacity(0.10),
            blurRadius: 6,
            offset: const Offset(0, 5),
          ),
        ],
      ), //The Design of the Product data within the page, arranged within a rectangular box for clear navigation.
      child: InkWell(
        borderRadius: BorderRadius.circular(10),
        onTap: () {
          Navigator.push(
            context,
            MaterialPageRoute(
              builder: (context) => ProductDetailPage(product: product),
            ),
          ); //supports navigating features when clicking the box, it provides an effects which the user had chosen the product.
        },
        child: ListTile(
          //arranging the text by rows and columns
          leading: ClipRRect(
            //Trimming the Edges of the Box
            borderRadius: BorderRadius.circular(10),
            child: Image.network(
              //allows to call the images from the external websites by the use of URL
              product.imagePath, //edits the image size that fits from the box
              width: 56,
              height: 56,
              fit: BoxFit.cover,
              semanticLabel: product.name, //edits the product name
            ),
          ),
          title: Text(product.name), //display the output of the sample product
          subtitle: Text(
            '₱${product.price.toStringAsFixed(2)} • ${product.category.label}',
          ), //display the price and category of the product
          trailing: const Icon(
            Icons.chevron_right,
          ), //Indicator where you can view the product, symbolizes the icon '>'
        ),
      ),
    );
  }
}

class ProductCard extends StatelessWidget {
  const ProductCard({super.key, required this.product});
  final Product product; //calling the required product for the displayed output

  @override
  Widget build(BuildContext context) {
    final colorScheme = Theme.of(
      context,
    ).colorScheme; //allows to edit the price color
    return InkWell(
      onTap: () {
        Navigator.push(
          context,
          MaterialPageRoute(
            builder: (context) => ProductDetailPage(product: product),
          ),
        ); //calling the product data and displaying for the output
      },
      child: Card(
        elevation: 4,
        margin: const EdgeInsets.all(4),
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
        clipBehavior: Clip.antiAlias,
        child: Column(
          //handles the design of the product card layout or the designed box
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            AspectRatio(
              aspectRatio: 1.2,
              child: Image.network(
                product.imagePath,
                fit: BoxFit.cover,
                semanticLabel: product.name,
              ),
            ),
            Padding(
              padding: const EdgeInsets.all(12),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    product.name,
                    style: Theme.of(context).textTheme.titleMedium,

              ),

      ),
    );
  }
}

class ProductDetailPage extends StatelessWidget {
  final Product product;
  const ProductDetailPage({super.key, required this.product});

  @override
  Widget build(BuildContext context) {
    final colorScheme = Theme.of(context).colorScheme;
    return Scaffold(
      //creates a layout for the whole page of the application with appbar and body for product details
      appBar: AppBar(title: Text(product.name)),
      body: SingleChildScrollView(
        child: Column(
          crossAxisAlignment: CrossAxisAlignment
              .start, //enables to format the prouct details from the page
          children: [
            Image.network(
              product.imagePath,
              width: double.infinity,
              height: 300,
              fit: BoxFit.cover,
              semanticLabel: product.name,
            ),
            Padding(
              padding: const EdgeInsets.all(16),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  //provide a layout for the product details such the name, price, and description
                  Row(
                    children: [
                      Chip(label: Text(product.category.label)),
                      const SizedBox(width: 8),
                      Chip(label: Text('${product.year}')),
                    ], //layout the product category and year in a one line row from the left siede
                  ),
                  const SizedBox(height: 12),
                  Text(
                    product.name,
                    style: Theme.of(context).textTheme.headlineSmall,
                  ),
                  const SizedBox(height: 8),
                  Text(
                    '₱${product.price.toStringAsFixed(2)}',
                    style: TextStyle(
                      fontSize: 22,
                      fontWeight: FontWeight.bold,
                      color: colorScheme.primary,
                    ),
                  ),
                  const SizedBox(height: 16),
                  Text(
                    product.description,
                    style: Theme.of(context).textTheme.bodyLarge,
                  ),
                ],
              ),
            ),
          ],
        ),
      ),
    );
  }
}

//The Whole Block Syntax Below are the mind of the whole application where able to navigate, select, or call the required product.
class CatalogPage extends StatefulWidget {
  const CatalogPage({super.key, required this.items});
  final List<Product> items;

  @override
  State<CatalogPage> createState() => _CatalogPageState();
}

class _CatalogPageState extends State<CatalogPage> {
  ProductCategory? _selected;

  @override
  Widget build(BuildContext context) {
    final filtered = _selected == null
        ? widget.items
        : widget.items.where((p) => p.category == _selected).toList();

    return Column(
      children: [
        SingleChildScrollView(
          scrollDirection: Axis.horizontal,
          padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 8),
          child: Row(
            children: [
              FilterChip(
                label: const Text('All'),
                selected: _selected == null,
                onSelected: (_) => setState(() => _selected = null),
              ),
              const SizedBox(width: 8),
              ...ProductCategory.values.map((cat) {
                return Padding(
                  padding: const EdgeInsets.only(right: 8),
                  child: FilterChip(
                    label: Text(cat.label),
                    selected: _selected == cat,
                    onSelected: (_) => setState(() => _selected = cat),
                  ),
                );
              }),
            ],
          ),
        ),
        Expanded(
          child: LayoutBuilder(
            builder: (context, constraints) {
              final w = constraints.maxWidth;
              if (filtered.isEmpty) {
                return const Center(
                  child: Text('No products in this category'),
                );
              }
              if (w < 600) {
                return ListView.builder(
                  itemCount: filtered.length,
                  itemBuilder: (_, i) => ProductTile(product: filtered[i]),
                );
              }
              return GridView.builder(
                padding: const EdgeInsets.all(12),
                gridDelegate: SliverGridDelegateWithFixedCrossAxisCount(
                  crossAxisCount: w < 840 ? 2 : 3,
                  mainAxisSpacing: 12,
                  crossAxisSpacing: 12,
                  childAspectRatio: 0.8,
                ),
                itemCount: filtered.length,
                itemBuilder: (_, i) => ProductCard(product: filtered[i]),
              );
            },
          ),
        ),
      ],
    );
  }
}
